# 企业垂直知识库 Agent 后台 —— 需求分析与设计

> 角色：项目经理 Agent（只做需求分析、数据库设计、接口规划，不含任何 Python 业务实现代码）
> 场景：政企运维，企业内部 PDF/Word/Excel 私有文档，自然语言问答，Agent 自动路由（RAG / 业务库查询 / 报表生成）

---

## 1. 完整 MySQL 建表 SQL

```sql
-- =====================================================================
-- 企业垂直知识库 Agent 后台  数据库初始化脚本
-- 目标库：MySQL 8.0+ / utf8mb4 / InnoDB
-- 说明：向量数据存 Chroma，MySQL 只存结构化元数据与业务数据
-- =====================================================================

CREATE DATABASE IF NOT EXISTS kb_agent
  DEFAULT CHARACTER SET utf8mb4
  DEFAULT COLLATE utf8mb4_unicode_ci;
USE kb_agent;

-- ---------------------------------------------------------------------
-- 1. 用户表（仅两种角色，简化 RBAC）
-- ---------------------------------------------------------------------
CREATE TABLE users (
  id            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  username      VARCHAR(64)     NOT NULL COMMENT '登录名',
  password_hash VARCHAR(255)    NOT NULL COMMENT 'bcrypt 密码哈希',
  display_name  VARCHAR(64)     DEFAULT NULL COMMENT '显示名',
  email         VARCHAR(128)    DEFAULT NULL COMMENT '邮箱',
  role          ENUM('admin','user') NOT NULL DEFAULT 'user' COMMENT '角色：admin=管理员 user=普通用户',
  status        ENUM('active','disabled') NOT NULL DEFAULT 'active' COMMENT '账号状态',
  created_at    DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at    DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uk_users_username (username),
  KEY idx_users_role (role),
  KEY idx_users_status (status)
) ENGINE=InnoDB COMMENT='用户表';

-- ---------------------------------------------------------------------
-- 2. 知识库表（知识库隔离的最小单元）
-- ---------------------------------------------------------------------
CREATE TABLE knowledge_bases (
  id               BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  name             VARCHAR(128)    NOT NULL COMMENT '知识库名称',
  description      VARCHAR(512)    DEFAULT NULL COMMENT '描述',
  owner_id         BIGINT UNSIGNED NOT NULL COMMENT '创建者(管理员) users.id',
  embedding_model  VARCHAR(64)     NOT NULL DEFAULT 'default' COMMENT 'Embedding 模型标识',
  chunk_size       INT             NOT NULL DEFAULT 500 COMMENT '分块大小(字符)',
  chunk_overlap    INT             NOT NULL DEFAULT 50 COMMENT '分块重叠(字符)',
  top_k            INT             NOT NULL DEFAULT 5 COMMENT '默认召回条数',
  rerank_threshold FLOAT           NOT NULL DEFAULT 0.0 COMMENT '重排序过滤阈值',
  status           ENUM('active','archived') NOT NULL DEFAULT 'active',
  doc_count        INT UNSIGNED    NOT NULL DEFAULT 0 COMMENT '文档数(冗余统计)',
  created_at       DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at       DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_kbs_owner (owner_id),
  KEY idx_kbs_status (status),
  CONSTRAINT fk_kbs_owner FOREIGN KEY (owner_id) REFERENCES users(id)
) ENGINE=InnoDB COMMENT='知识库表';

-- ---------------------------------------------------------------------
-- 3. 知识库-用户权限表（简化 RBAC 的知识库隔离授权）
-- ---------------------------------------------------------------------
CREATE TABLE kb_permissions (
  id         BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  kb_id      BIGINT UNSIGNED NOT NULL COMMENT '知识库ID',
  user_id    BIGINT UNSIGNED NOT NULL COMMENT '被授权用户ID',
  permission ENUM('read','write') NOT NULL DEFAULT 'read' COMMENT 'read=可问答/只读 write=可上传管理文档',
  granted_by BIGINT UNSIGNED DEFAULT NULL COMMENT '授权人 users.id',
  created_at DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uk_kb_user (kb_id, user_id),
  KEY idx_kbperm_user (user_id),
  CONSTRAINT fk_kbperm_kb   FOREIGN KEY (kb_id)   REFERENCES knowledge_bases(id) ON DELETE CASCADE,
  CONSTRAINT fk_kbperm_user FOREIGN KEY (user_id) REFERENCES users(id)          ON DELETE CASCADE
) ENGINE=InnoDB COMMENT='知识库访问授权（admin 默认全库可访问，无需授权记录）';

-- ---------------------------------------------------------------------
-- 4. 文档表（上传的私有文档）
-- ---------------------------------------------------------------------
CREATE TABLE documents (
  id           BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  kb_id        BIGINT UNSIGNED NOT NULL COMMENT '所属知识库',
  title        VARCHAR(255)    NOT NULL COMMENT '文档标题/文件名',
  file_type    ENUM('pdf','docx','xlsx','md','txt') NOT NULL COMMENT '文件类型',
  file_path    VARCHAR(512)    NOT NULL COMMENT '对象存储/本地路径',
  file_size    BIGINT UNSIGNED NOT NULL DEFAULT 0 COMMENT '字节数',
  file_hash    CHAR(64)        DEFAULT NULL COMMENT 'SHA256，用于去重',
  parse_status ENUM('pending','parsing','chunking','embedding','done','failed') NOT NULL DEFAULT 'pending',
  chunk_count  INT UNSIGNED    NOT NULL DEFAULT 0 COMMENT '已生成分块数',
  error_msg    VARCHAR(512)    DEFAULT NULL COMMENT '失败原因',
  uploaded_by  BIGINT UNSIGNED NOT NULL COMMENT '上传人 users.id',
  created_at   DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at   DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_docs_kb (kb_id),
  KEY idx_docs_status (parse_status),
  KEY idx_docs_hash (file_hash),
  CONSTRAINT fk_docs_kb FOREIGN KEY (kb_id) REFERENCES knowledge_bases(id) ON DELETE CASCADE
) ENGINE=InnoDB COMMENT='文档表';

-- ---------------------------------------------------------------------
-- 5. 文档分块表（RAG 文本块；向量本体存 Chroma）
-- ---------------------------------------------------------------------
CREATE TABLE document_chunks (
  id           BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  doc_id       BIGINT UNSIGNED NOT NULL COMMENT '所属文档',
  kb_id        BIGINT UNSIGNED NOT NULL COMMENT '所属知识库(冗余便于过滤)',
  chunk_index  INT UNSIGNED    NOT NULL COMMENT '块序号',
  content      MEDIUMTEXT      NOT NULL COMMENT '块原文',
  token_count  INT UNSIGNED    NOT NULL DEFAULT 0 COMMENT '估算 token 数',
  metadata     JSON            DEFAULT NULL COMMENT '页码/标题/行列等来源信息',
  chroma_id    VARCHAR(128)    DEFAULT NULL COMMENT 'Chroma 中向量ID',
  created_at   DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uk_chunk_chroma (chroma_id),
  KEY idx_chunks_doc (doc_id),
  KEY idx_chunks_kb (kb_id),
  FULLTEXT KEY ft_chunks_content (content) WITH PARSER ngram COMMENT 'BM25 备选/混合检索用',
  CONSTRAINT fk_chunks_doc FOREIGN KEY (doc_id) REFERENCES documents(id) ON DELETE CASCADE,
  CONSTRAINT fk_chunks_kb  FOREIGN KEY (kb_id)  REFERENCES knowledge_bases(id) ON DELETE CASCADE
) ENGINE=InnoDB COMMENT='文档分块表';

-- ---------------------------------------------------------------------
-- 6. 会话表
-- ---------------------------------------------------------------------
CREATE TABLE conversations (
  id         BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  user_id    BIGINT UNSIGNED NOT NULL COMMENT '发起人',
  kb_id      BIGINT UNSIGNED DEFAULT NULL COMMENT '绑定的主知识库(可空)',
  title      VARCHAR(255)    DEFAULT NULL COMMENT '会话标题(首问摘要)',
  created_at DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
  updated_at DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_conv_user (user_id),
  KEY idx_conv_kb (kb_id),
  CONSTRAINT fk_conv_user FOREIGN KEY (user_id) REFERENCES users(id) ON DELETE CASCADE
) ENGINE=InnoDB COMMENT='对话会话表';

-- ---------------------------------------------------------------------
-- 7. 消息表（单轮问答 / 多轮消息）
-- ---------------------------------------------------------------------
CREATE TABLE messages (
  id              BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  conversation_id BIGINT UNSIGNED NOT NULL COMMENT '所属会话',
  user_id         BIGINT UNSIGNED NOT NULL COMMENT '发言用户',
  role            ENUM('user','assistant','system','tool') NOT NULL COMMENT '消息角色',
  content         MEDIUMTEXT      NOT NULL COMMENT '消息正文',
  agent_route     ENUM('rag','db_query','report','direct') DEFAULT NULL COMMENT 'Agent 最终路由',
  trace_id        CHAR(36)        DEFAULT NULL COMMENT '关联 Trace',
  input_tokens    INT UNSIGNED    NOT NULL DEFAULT 0,
  output_tokens   INT UNSIGNED    NOT NULL DEFAULT 0,
  latency_ms      INT UNSIGNED    NOT NULL DEFAULT 0 COMMENT '本消息总耗时(ms)',
  status          ENUM('running','done','failed') NOT NULL DEFAULT 'running',
  error_msg       VARCHAR(512)    DEFAULT NULL,
  created_at      DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_msg_conv (conversation_id),
  KEY idx_msg_user (user_id),
  KEY idx_msg_trace (trace_id),
  KEY idx_msg_route (agent_route),
  CONSTRAINT fk_msg_conv FOREIGN KEY (conversation_id) REFERENCES conversations(id) ON DELETE CASCADE
) ENGINE=InnoDB COMMENT='消息表';

-- ---------------------------------------------------------------------
-- 8. Trace 调用日志表（自建，记录 token / 耗时 / 路由 / 检索统计）
-- ---------------------------------------------------------------------
CREATE TABLE trace_logs (
  id              BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  trace_id        CHAR(36)        NOT NULL COMMENT '链路ID(UUID)',
  conversation_id BIGINT UNSIGNED DEFAULT NULL,
  message_id      BIGINT UNSIGNED DEFAULT NULL,
  user_id         BIGINT UNSIGNED DEFAULT NULL,
  kb_id           BIGINT UNSIGNED DEFAULT NULL COMMENT '命中的知识库',
  session_id      VARCHAR(64)     DEFAULT NULL COMMENT '运行会话ID(可选)',
  agent_route     ENUM('rag','db_query','report','direct') DEFAULT NULL COMMENT '最终路由决策',
  model           VARCHAR(64)     NOT NULL DEFAULT 'claude-3-5-sonnet-20241022',
  input_tokens    INT UNSIGNED    NOT NULL DEFAULT 0,
  output_tokens   INT UNSIGNED    NOT NULL DEFAULT 0,
  total_tokens    INT UNSIGNED    NOT NULL DEFAULT 0,
  llm_call_count  INT UNSIGNED    NOT NULL DEFAULT 0 COMMENT 'LLM 调用次数',
  tool_call_count INT UNSIGNED    NOT NULL DEFAULT 0 COMMENT '工具调用次数',
  num_chunks      INT UNSIGNED    NOT NULL DEFAULT 0 COMMENT '检索召回块数',
  retrieval_ms    INT UNSIGNED    NOT NULL DEFAULT 0 COMMENT '检索耗时(ms)',
  total_latency_ms INT UNSIGNED   NOT NULL DEFAULT 0 COMMENT '端到端总耗时(ms)',
  status          ENUM('running','success','failed') NOT NULL DEFAULT 'running',
  error_msg       VARCHAR(1024)   DEFAULT NULL,
  started_at      DATETIME        DEFAULT NULL,
  ended_at        DATETIME        DEFAULT NULL,
  created_at      DATETIME        NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uk_trace_traceid (trace_id),
  KEY idx_trace_user (user_id),
  KEY idx_trace_route (agent_route),
  KEY idx_trace_status (status),
  KEY idx_trace_created (created_at),
```

## 2. REST 接口清单

> 统一约定：
> - 前缀：`/api/v1`；认证：除 `POST /auth/login` 与 `GET /health` 外均需 `Authorization: Bearer <JWT>`。
> - 统一响应包：`{ "code": 0, "message": "ok", "data": {...} }`；分页统一 `{ "total": n, "items": [...] }`。
> - 访问权限：`admin` = 仅管理员；`user` = 登录用户（受知识库授权约束）；`kb_read` = 对该知识库有 read/write 权限；`kb_write` = 对该知识库有 write 权限。
> - 流式问答使用 `POST /chat/stream`（SSE），非流式使用 `POST /chat`。

### 2.1 认证与用户

| 路径 | 方法 | 入参 | 返回 | 权限 |
|---|---|---|---|---|
| `/auth/login` | POST | `{username, password}` | `{access_token, refresh_token, user}` | 公开 |
| `/auth/refresh` | POST | `{refresh_token}` | `{access_token}` | 公开 |
| `/auth/me` | GET | — | `user 对象` | user |
| `/auth/logout` | POST | — | `{}` | user |
| `/auth/change-password` | POST | `{old_password, new_password}` | `{}` | user |
| `/users` | GET | `page, size, keyword, role` | 分页用户列表 | admin |
| `/users` | POST | `{username, password, display_name, email, role}` | 新建用户 | admin |
| `/users/{id}` | PUT | `{display_name, email, role, status}` | 更新后用户 | admin |
| `/users/{id}` | DELETE | — | `{}`（禁用/删除） | admin |
| `/users/{id}/reset-password` | PUT | `{new_password}` | `{}` | admin |

### 2.2 知识库管理

| 路径 | 方法 | 入参 | 返回 | 权限 |
|---|---|---|---|---|
| `/kbs` | GET | `page, size, keyword` | 分页（普通用户仅返回已授权库） | user |
| `/kbs` | POST | `{name, description, chunk_size, chunk_overlap, top_k, rerank_threshold}` | 新建知识库 | admin |
| `/kbs/{id}` | GET | — | 知识库详情 | kb_read/admin |
| `/kbs/{id}` | PUT | 同 POST 可更新字段 | 更新后知识库 | admin |
| `/kbs/{id}` | DELETE | — | `{}`（级联删除文档/分块/向量） | admin |

## 4. 各模块验收标准

### 4.1 用户与认证模块
- [ ] 仅存在 `admin` / `user` 两种角色；`role` 字段受限，无法通过接口越权写入其他角色。
- [ ] 登录成功签发 JWT，携带 `user_id`/`role`；过期或篡改 Token 返回 401。
- [ ] 普通用户只能访问被 `kb_permissions` 授权（read/write）的知识库，未授权库一律不可见、不可问答。
- [ ] `admin` 默认可管理全部知识库与用户；普通用户访问管理接口（用户管理/评测/Trace）一律 403。
- [ ] 密码以 bcrypt 哈希存储，数据库中不存在明文；关键操作写入 `operation_logs`。

### 4.2 知识库管理模块
- [ ] 管理员可创建/编辑/删除知识库；删除时级联清理文档、分块、Chroma 向量与授权记录。
- [ ] 支持配置分块大小、重叠、召回 top_k、重排序阈值，配置即时生效。
- [ ] 授权接口可对指定用户授予 read/write，重复授权幂等，撤销后该用户立即失去访问。
- [ ] `stats` 接口返回的文档数、分块数、token 总量与库内实际一致。

### 4.3 文档接入与 RAG 检索模块
- [ ] 支持 PDF / Word / Excel 上传，单文件解析成功率 ≥ 95%（对可解析样本集）。
- [ ] 解析→分块→向量化全链路异步完成，`parse_status` 正确流转到 `done`/`failed` 并记录失败原因。
- [ ] 相同 `file_hash` 的重复上传被识别并去重。
- [ ] 混合检索同时返回向量命中和 BM25 命中的并集，并经重排序输出 top_k，端到端检索耗时 P95 < 2s（普通规模 ≤ 10 万块）。
- [ ] 删除文档后，Chroma 中对应向量与 BM25 索引同步清除，无脏数据。

### 4.4 智能问答 Agent 模块（LangGraph 三工具）
- [ ] LangGraph 包含 3 个工具节点：`rag_tool`（知识库检索）、`db_query_tool`（业务库查询）、`report_tool`（Markdown 报表生成），并由 router 节点正确选择。
- [ ] 意图路由判定：知识类问题→RAG，台账/工单/设备/告警等结构化查询→业务库，汇总/统计/周报类→报表，与用例集匹配准确率 ≥ 85%。
- [ ] 回答附带引用来源（chunk / 文档标题），无依据时不编造（明确表示未检索到）。
- [ ] 支持多轮上下文与流式(SSE)输出；一次问答只调用必要工具，不出现无效重复调用。
- [ ] 业务库查询基于 `biz_devices`/`biz_tickets`/`biz_alerts` 返回正确结构化结果；报表输出为标准 Markdown，可下载为 `.md` 文件。

### 4.5 Trace 日志模块
- [ ] 每次问答生成唯一 `trace_id`，100% 覆盖记录：路由结果、LLM 调用次数、输入/输出 token、总耗时、检索耗时与召回块数。
- [ ] 每次工具调用在 `trace_tool_calls` 中留一条明细，含入参、输出、token、耗时与状态。
- [ ] Trace 详情页可完整还原一次请求的“路由→工具→最终回答”调用链。
- [ ] 汇总接口可统计调用量、平均耗时、token 消耗与路由分布，且数值与明细一致。
- [ ] Trace 写入不阻塞主问答流程（异常时不影响回答返回）。

### 4.6 RAGAS 评测模块
- [ ] 管理员可创建数据集并批量导入（问题+参考答案+参考上下文），字段校验通过。
- [ ] 支持对指定知识库发起评测，逐条产出 `faithfulness` / `answer_relevancy` / `context_precision` / `context_recall` 四项得分。
- [ ] 运行结束汇总均值写入 `eval_runs.metrics`，评测状态正确流转（pending→running→done/failed）。
- [ ] 提供基准验收：在默认测试集上 `faithfulness ≥ 0.7`、`answer_relevancy ≥ 0.7`、`context_recall ≥ 0.7` 方可视为通过。
- [ ] 单条得分可下钻查看生成答案与召回上下文，便于定位问题。

### 4.7 Streamlit 演示模块
- [ ] 提供登录页，能区分管理员与普通用户，按角色显示对应菜单。
- [ ] 知识库管理页：支持上传文档、查看解析状态、查看文档/分块列表。
- [ ] 智能问答页：支持选择知识库、多轮对话、展示答案来源与路由标签。
- [ ] Trace 日志页（管理员）：可检索/筛选并按调用链查看明细。
- [ ] RAGAS 评测页（管理员）：可创建数据集、发起评测、查看四项指标得分。

### 4.8 部署与运维模块
- [ ] `docker-compose up` 一键启动 MySQL、Redis、Chroma 与应用，健康检查 `/health` 三组件全绿。
- [ ] 环境变量（数据库连接、模型密钥、Chroma 地址等）通过 `.env` 注入，不硬编码于代码。
- [ ] 后端镜像可独立构建（`docker build`）并支持重启后自动迁移建表。
- [ ] 提供 README，包含启动步骤、默认账号、接口文档入口与演示指引。
